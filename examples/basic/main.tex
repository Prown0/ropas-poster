          \blocktitle{1. Mathematical expressions}

          If \(x\leq y\) and \(y\leq z\), then \(x\leq z\) for all \(x,y,z\in X\).
          That is, \[\forall x,y,z\in X.\,x\leq y\mathrel{\land}y\leq z\rightarrow x\leq z.\]

        \begin{posterblock}{2}{1}
          \blocktitle{2. Lists}

          Let \(G\) be a set, \({\bullet}:G\times G\mathbin{\rightarrow}G\), \((\cdot)^{-1}:G\mathbin{\rightarrow}G\), and \(e\in G\).
          If \begin{itemize}
            \item
            \(\bullet\) is associative,
            \item
            \(e\bullet x=x\) and \(x\bullet e=x\) for all \(x\in G\), and
            \item
            \(x^{-1}\bullet x=e\) and \(x\bullet x^{-1}=e\) for all \(x\in G\),
          \end{itemize} then we call \(\left(G,{\bullet},(\cdot)^{-1},e\right)\) a \textbf{group}.

          Let \(G\) be a set and \({\bullet}:G\times G\mathbin{\rightarrow}G\).
          If \begin{enumerate}
            \item
            \(\bullet\) is associative, and
            \item
            there exists \(e\in G\) such that \begin{enumerate}
              \item
              \(e\bullet x=x\), and
              \item
              there exists \(y\in G\) such that \(y\bullet x=e\) for all \(x\in G\),
            \end{enumerate} then there uniquely exists a group with \(G\) and \(\bullet\).
          \end{enumerate}
        \end{posterblock}
